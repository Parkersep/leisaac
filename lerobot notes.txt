from pxr import UsdPhysics, Gf
import math

# Define your known SO101 limits (in degrees) from your configuration
so101_limits = {
    "Rotation": (-110.0, 110.0),    # shoulder_pan
    "Pitch": (-100.0, 100.0),       # shoulder_lift
    "Elbow": (-100.0, 90.0),        # elbow_flex
    "Wrist_Pitch": (-95.0, 95.0),   # wrist_flex
    "Wrist_Roll": (-160.0, 160.0),  # wrist_roll
    "Jaw": (-10.0, 100.0)           # gripper
}

def sync_usd_limits(robot_root_path="/World/envs/env_0/Robot"):
    """
    Overwrites USD revolute joint limits to match SO101 hardware specs.
    Processes both the primary arm and the '_2' suffixed joints.
    """
    stage = omni.usd.get_context().get_stage()
    
    print(f"--- Updating Joint Limits for {robot_root_path} ---")

    for joint_base_name, (low_deg, high_deg) in so101_limits.items():
        # Handle both arms by checking for the base name and the suffix
        for suffix in ["", "_2"]:
            joint_name = f"{joint_base_name}{suffix}"
            
            # Search the stage for the joint prim
            # Based on your image, they may be under a 'joints' scope or nested
            found_prim = None
            for prim in stage.Traverse():
                if prim.GetName() == joint_name and prim.IsA(UsdPhysics.RevoluteJoint):
                    found_prim = prim
                    break
            
            if found_prim:
                joint = UsdPhysics.RevoluteJoint(found_prim)
                
                # Overwrite limits (Isaac Sim stores these as degrees in the backend for RevoluteJoint)
                joint.GetLowerLimitAttr().Set(low_deg)
                joint.GetUpperLimitAttr().Set(high_deg)
                
                print(f"SUCCESS: {joint_name} set to ({low_deg}, {high_deg}) degrees.")
            else:
                print(f"WARNING: Joint '{joint_name}' not found in the stage.")

# Run the update
sync_usd_limits()
